# 💡 Solución

---
Primero se hace el select para confirmar el numero de rollos, luego se eliminar y a las finales se deja pasar 
```sql
begin tran
--select * into TX_ORDTRA_ITEMS_ROLLOS_46499  from TX_ORDTRA_ITEMS_ROLLOS where Cod_Ordtra = '46499' and Num_Secuencia = '1'
select * from TX_ORDTRA_ITEMS_ROLLOS where Cod_Ordtra = '47045' and Num_Secuencia = '1'

delete from TX_ORDTRA_ITEMS_ROLLOS where Cod_Ordtra = '47045' and Num_Secuencia = '1'
EXEC UP_ELIMINA_ROLLO
 @COD_TIPORDTRA    = 'TI'
,@COD_ORDTRA       = '47045      '
,@NUM_SECUENCIA_ORDEN  =  1
,@NUM_ROLLO        =  1
,@Cod_Usuario      = 'cgalam'

rollback tran
```
